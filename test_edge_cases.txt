## start[0] = 0       # zero register should always be 0
## start[1] = 100
## start[2] = -100

# Test zero register cannot be overwritten
addi x0, x1, 50      # Should not change x0
add  x3, x0, x1      # x3 = 0 + 100 = 100

# Test negative numbers
addi x4, x2, 50      # x4 = -100 + 50 = -50
slti x5, x2, 0       # x5 = -100 < 0 = 1

# Test large shifts
addi x6, x0, 1
sll  x7, x6, 31      # x7 = 1 << 31 (should be negative in signed)

# Test memory alignment
addi x8, x0, 0x12345678
sw   x8, 0(x0)       # Store at address 0
lw   x9, 0(x0)       # Load back
